# Supabase Kurulum — 6. Aşama: IP bazlı hız sınırı

Aynı IP adresinden **1 saatte en fazla 15 kayıt**. Hem dilekçe hem gönüllü kaydı için geçerli.

**Nereye:** `https://supabase.com/dashboard/project/mednnlpprdgupdajmrin/sql/new`

---

## Tasarım notları

**Ham IP saklanmıyor.** IP adresi KVKK kapsamında kişisel veridir. Sunucuda tutulan gizli bir
pepper ile SHA-256'lanıp yalnızca hash saklanıyor. Veritabanı dökümü alınsa bile IP'ler geri
elde edilemez (pepper da DB'de ama tablo anon'a kapalı ve hash'ler 2 saat sonra siliniyor).

**Eski kayıtlar otomatik temizleniyor.** Her çağrıda 2 saatten eski satırlar siliniyor;
tablo şişmiyor, ayrı bir cron gerekmiyor.

**IP tespit edilemezse sınır uygulanmaz.** PostgREST başlıkları bazı durumlarda gelmeyebilir.
O durumda kayıt kabul edilir — kullanıcıyı engellemektense sınırı atlamayı tercih ettim.
Aşağıdaki `rate_limit_probe()` ile tespitin çalıştığını doğrulayacağız.

**Paylaşılan IP uyarısı:** Kurumsal ofis, üniversite, kafe wifi'si veya mobil operatör CGNAT
arkasındaki kullanıcılar aynı IP'yi paylaşır. Bir kampanya etkinliğinde aynı mekândan 15'ten
fazla kişi imza verirse 16.'sı engellenir. Sayıyı yükseltmek için scriptin sonundaki nota bakın.

---

## Beklenen sonuç

Son sorgu IP tespitinin çalıştığını gösterir:

| ip_tespit_edildi | maskeli_ip | bulunan_baslik |
|---|---|---|
| `true` | `85.104...` | `x-forwarded-for` |

`ip_tespit_edildi = false` gelirse bana söyleyin — başlık adını değiştirmem gerekir.

---

```sql
-- =====================================================================
-- 7527iptal.org - IP bazli hiz siniri
-- Tekrar calistirilabilir.
-- =====================================================================

create extension if not exists pgcrypto with schema extensions;

-- ---------------------------------------------------------------------
-- 1) Sunucu tarafi gizli anahtar (IP hash'lemek icin)
-- ---------------------------------------------------------------------
create table if not exists public.app_secrets (
  name       text primary key,
  value      text not null,
  created_at timestamptz not null default now()
);

alter table public.app_secrets enable row level security;
revoke all on public.app_secrets from anon, authenticated;

insert into public.app_secrets (name, value)
values ('ip_pepper', encode(extensions.gen_random_bytes(32), 'hex'))
on conflict (name) do nothing;


-- ---------------------------------------------------------------------
-- 2) Hiz siniri kayit tablosu  (yalnizca hash + zaman)
-- ---------------------------------------------------------------------
create table if not exists public.submission_rate (
  id         bigint generated by default as identity primary key,
  scope      text not null,
  ip_hash    text not null,
  created_at timestamptz not null default now()
);

create index if not exists submission_rate_lookup_idx
  on public.submission_rate (scope, ip_hash, created_at desc);

alter table public.submission_rate enable row level security;
revoke all on public.submission_rate from anon, authenticated;


-- ---------------------------------------------------------------------
-- 3) Istemci IP'sini basliklardan cikar
-- ---------------------------------------------------------------------
create or replace function public.client_ip()
returns text
language plpgsql
stable
security definer
set search_path = public
as $$
declare
  hdrs json;
  ip   text;
begin
  begin
    hdrs := current_setting('request.headers', true)::json;
  exception when others then
    return null;
  end;

  if hdrs is null then
    return null;
  end if;

  ip := nullif(btrim(coalesce(hdrs ->> 'cf-connecting-ip', '')), '');
  if ip is null then
    ip := nullif(btrim(split_part(coalesce(hdrs ->> 'x-forwarded-for', ''), ',', 1)), '');
  end if;
  if ip is null then
    ip := nullif(btrim(coalesce(hdrs ->> 'x-real-ip', '')), '');
  end if;

  return ip;
end;
$$;

revoke all on function public.client_ip() from public;


-- ---------------------------------------------------------------------
-- 4) Hiz siniri kontrolu
--    Sinir asilirsa 'rate_limited' firlatir.
-- ---------------------------------------------------------------------
create or replace function public.check_rate_limit(p_scope text, p_max integer)
returns void
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_ip     text;
  v_pepper text;
  v_hash   text;
  v_count  integer;
begin
  v_ip := public.client_ip();

  -- IP tespit edilemediyse sinir uygulanmaz (kullaniciyi engelleme)
  if v_ip is null then
    return;
  end if;

  select value into v_pepper from public.app_secrets where name = 'ip_pepper';
  if v_pepper is null then
    return;
  end if;

  v_hash := encode(extensions.digest(v_ip || ':' || v_pepper, 'sha256'), 'hex');

  -- Eski kayitlari firsat buldukca temizle
  delete from public.submission_rate
  where created_at < now() - interval '2 hours';

  select count(*) into v_count
  from public.submission_rate
  where scope = p_scope
    and ip_hash = v_hash
    and created_at > now() - interval '1 hour';

  if v_count >= p_max then
    raise exception 'rate_limited';
  end if;

  insert into public.submission_rate (scope, ip_hash) values (p_scope, v_hash);
end;
$$;

revoke all on function public.check_rate_limit(text, integer) from public;


-- ---------------------------------------------------------------------
-- 5) submit_petition -> hiz siniri ekle
-- ---------------------------------------------------------------------
create or replace function public.submit_petition(
  p_recipient_type text,
  p_recipient_id   integer,
  p_payload_enc    text
)
returns text
language plpgsql
security definer
set search_path = public
as $$
declare
  v_number text;
  v_id     bigint;
begin
  if p_recipient_type not in ('cb', 'mb', 'mp') then
    raise exception 'invalid_recipient';
  end if;

  if p_payload_enc is null
     or length(p_payload_enc) < 64
     or length(p_payload_enc) > 4000000
     or p_payload_enc not like '{"v":1%' then
    raise exception 'invalid_payload';
  end if;

  -- Ayni IP'den saatte en fazla 15 dilekce
  perform public.check_rate_limit('petition', 15);

  v_number := '7527-' || upper(p_recipient_type) || '-' ||
              to_char(now() at time zone 'Europe/Istanbul', 'YYYYMMDD') || '-' ||
              lpad(nextval('public.petition_seq')::text, 16, '0');

  insert into public.campaign_submissions
    (recipient_type, recipient_id, petition_number, payload_enc)
  values
    (p_recipient_type, p_recipient_id, v_number, p_payload_enc)
  returning id into v_id;

  if to_regclass('public.campaign_submission_audit') is not null then
    insert into public.campaign_submission_audit (submission_id, action)
    values (v_id, 'created');
  end if;

  return v_number;
end;
$$;

revoke all on function public.submit_petition(text, integer, text) from public;
grant execute on function public.submit_petition(text, integer, text) to anon, authenticated;


-- ---------------------------------------------------------------------
-- 6) add_volunteer_enc -> hiz siniri ekle
-- ---------------------------------------------------------------------
create or replace function public.add_volunteer_enc(
  p_city        text,
  p_payload_enc text
)
returns text
language plpgsql
security definer
set search_path = public
as $$
declare
  v_city text := nullif(btrim(coalesce(p_city, '')), '');
begin
  if v_city is not null and length(v_city) > 60 then
    raise exception 'invalid_city';
  end if;

  if p_payload_enc is null
     or length(p_payload_enc) < 64
     or length(p_payload_enc) > 1000000
     or p_payload_enc not like '{"v":1%' then
    raise exception 'invalid_payload';
  end if;

  perform public.check_rate_limit('volunteer', 15);

  insert into public.campaign_volunteers (city, payload_enc)
  values (v_city, p_payload_enc);

  return 'created';
end;
$$;

revoke all on function public.add_volunteer_enc(text, text) from public;
grant execute on function public.add_volunteer_enc(text, text) to anon, authenticated;


-- ---------------------------------------------------------------------
-- 7) Tani fonksiyonu  --  IP tespitinin calistigini dogrulamak icin
--    Ham IP'yi disariya vermez, yalnizca maskeli ilk iki oktet.
--    Dogrulama bittikten sonra silinebilir.
-- ---------------------------------------------------------------------
create or replace function public.rate_limit_probe()
returns table (
  ip_tespit_edildi boolean,
  maskeli_ip       text,
  bulunan_baslik   text
)
language plpgsql
security definer
set search_path = public
as $$
declare
  hdrs json;
  ip   text;
  src  text;
begin
  begin
    hdrs := current_setting('request.headers', true)::json;
  exception when others then
    hdrs := null;
  end;

  if hdrs is not null then
    if nullif(btrim(coalesce(hdrs ->> 'cf-connecting-ip', '')), '') is not null then
      src := 'cf-connecting-ip';
    elsif nullif(btrim(coalesce(hdrs ->> 'x-forwarded-for', '')), '') is not null then
      src := 'x-forwarded-for';
    elsif nullif(btrim(coalesce(hdrs ->> 'x-real-ip', '')), '') is not null then
      src := 'x-real-ip';
    end if;
  end if;

  ip := public.client_ip();

  return query select
    (ip is not null),
    case when ip is null then null
         else split_part(ip, '.', 1) || '.' || split_part(ip, '.', 2) || '...' end,
    src;
end;
$$;

revoke all on function public.rate_limit_probe() from public;
grant execute on function public.rate_limit_probe() to anon, authenticated;


-- ---------------------------------------------------------------------
-- 8) Yetkileri tekrar geri al
-- ---------------------------------------------------------------------
revoke all on public.app_secrets     from anon, authenticated;
revoke all on public.submission_rate from anon, authenticated;


-- =====================================================================
-- DOGRULAMA
-- =====================================================================
select * from public.rate_limit_probe();
-- ip_tespit_edildi = true olmali.
-- false gelirse haber verin; baslik adini degistirmem gerekir.
```

---

## Sınırı değiştirmek

Sayı iki yerde geçiyor — `submit_petition` içinde `check_rate_limit('petition', 15)` ve
`add_volunteer_enc` içinde `check_rate_limit('volunteer', 15)`. Değiştirmek için ilgili
fonksiyonu `create or replace` ile yeniden çalıştırmanız yeterli.

Kampanya günü aynı mekândan toplu imza toplayacaksanız (stand, etkinlik) bu sayıyı geçici
olarak yükseltmek isteyebilirsiniz.

## Sınıra takılan kullanıcı ne görür

Sunucu `rate_limited` hatası döndürür. `index.html` tarafında bunu
"Bu bağlantıdan kısa sürede çok fazla kayıt yapıldı. Lütfen bir saat sonra tekrar deneyin."
mesajına çevireceğim.
